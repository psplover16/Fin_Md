```sql
DELIMITER //

CREATE PROCEDURE 模擬雙11結帳(
    IN p_會員ID INT,
    IN p_新地址 VARCHAR(100),
    IN p_原地址版本號 INT,      -- 前端表單帶過來的舊版本號
    IN p_商品ID INT,
    IN p_購買數量 INT
)
BEGIN
    -- ==================================================
    -- 1. 區域變數 (DECLARE)：生命週期僅限 BEGIN...END 區塊內
    -- ==================================================
    DECLARE v_影響行數 INT DEFAULT 0;   -- 必須嚴格定義型別 INT
    DECLARE v_目前庫存 INT;
    
    -- 【交易控制進階技巧】：宣告發生 SQL 系統錯誤時的逃生路線
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;  -- 發生異常立刻倒退，確保資料一致性
        SET @最後狀態 = '失敗：系統發生未預期異常，已復原'; 
    END;

    -- ==================================================
    -- 2. 啟動交易 (Transaction Control)
    -- ==================================================
    START TRANSACTION;

    -- ==================================================
    -- 3. 樂觀鎖 (Optimistic Lock) 實作：修改收件地址
    -- ==================================================
    -- 情境：修改地址衝突率低，不鎖定資料，事後核對版本號
    UPDATE 會員表
    SET 地址 = p_新地址,
        version = version + 1           -- 成功修改的話，版本號自動 +1
    WHERE 會員ID = p_會員ID
      AND version = p_原地址版本號;       -- 【樂觀鎖核心】：核對版本號有沒有被別人偷改

    -- ROW_COUNT() 會回傳剛才 UPDATE 成功修改了幾筆
    -- v_影響行數 設定為 ROW_COUNT()
    SELECT ROW_COUNT() INTO v_影響行數;

    IF v_影響行數 = 0 THEN
        -- 影響行數為 0，代表 WHERE 條件沒命中，有人搶先改了這個會員的地址
        ROLLBACK;  -- 【交易控制】：放棄剛才的操作
        -- 使用者自訂變數 (@)：跨區塊存活，用來把結果傳遞到外部
        SET @最後狀態 = '失敗：地址已被其他裝置修改，請重新整理 (樂觀鎖攔截)';
        
    ELSE
        -- ==================================================
        -- 4. 悲觀鎖 (Pessimistic Lock) 實作：扣減庫存
        -- ==================================================
        -- 情境：限量庫存搶購，衝突率極高，直接鎖進 VIP 包廂
        
        -- 【悲觀鎖核心】：加上 FOR UPDATE，獨佔這筆商品的讀寫權
        SELECT 庫存量 INTO v_目前庫存
        FROM 商品表
        WHERE 商品ID = p_商品ID
        FOR UPDATE; 

        -- 檢查庫存夠不夠
        IF v_目前庫存 >= p_購買數量 THEN
            -- 庫存足夠，安心扣減 (此時沒有任何人能干擾這筆庫存資料)
            UPDATE 商品表
            SET 庫存量 = 庫存量 - p_購買數量
            WHERE 商品ID = p_商品ID;

            -- ==================================================
            -- 5. 提交交易 (Transaction Control)
            -- ==================================================
            COMMIT; -- 確定所有動作都沒問題，正式寫入硬碟並釋放排他鎖 (X-Lock)
            SET @最後狀態 = '成功：結帳完成，已成功搶到商品！';
            
        ELSE
            -- 庫存不足
            ROLLBACK; -- 【交易控制】：放棄地址修改與庫存操作
            SET @最後狀態 = '失敗：庫存不足，商品已被搶空';
        END IF;
    END IF;

END //
DELIMITER ;
```
